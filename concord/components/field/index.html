Form Field Group

A semantic fieldset for related controls with shared labeling, supporting description, and group-level validation.

Component implementation

packages/css/src/components/field.css 
.dsl-field,
.dsl-field-group {
  display: inline-block;
  inline-size: var(--dsl-field-width, 100%);
  margin: 0;
  padding: 0;
  border: 0;

  > :not(:last-child) {
    margin-block-end: var(--dsl-component-field-spacing-gap);
  }

  > .description,
  > .message {
    display: block;
    color: var(--dsl-component-field-content-color-description);
    font-size: var(--dsl-component-field-content-typography-size);
    line-height: var(--dsl-component-field-content-typography-line-height);
    overflow-wrap: anywhere;
  }

  > .message {
    color: var(--dsl-component-field-content-color-error);
  }
}

/* Concord Form Field composition contract v1.0.0
 *
 * Form Field is a composition contract, not another form control. The named
 * regions create predictable normal block flow around inline/inline-flex
 * components without changing the nested component's intrinsic sizing.
 */
.dsl-field {
  > .label,
  > .control {
    display: block;
    min-inline-size: 0;
  }
}

.dsl-field-group {
  > legend {
    padding: 0;
    color: var(--dsl-component-field-content-color-label);
    font-weight: var(--dsl-component-field-content-typography-weight);
  }

  > .fields {
    padding: var(--dsl-component-field-spacing-group-padding);
    border: var(--dsl-component-field-border-width) solid transparent;
    border-radius: var(--dsl-component-field-border-radius);
  }

  > .fields > :not(:last-child) {
    margin-block-end: var(--dsl-component-field-spacing-gap);
  }

  &[data-invalid] > .fields {
    border-color: var(--dsl-component-field-border-color-invalid);
  }
}

@media (forced-colors: active) {
  .dsl-field,
  .dsl-field-group {
    color: CanvasText;

    > .description,
    > .message,
    > legend {
      color: CanvasText;
    }
  }

  .dsl-field-group[data-invalid] > .fields {
    border-color: Mark;
  }
}
