.dsl-textarea {
  --_dsl-textarea-background: var(
    --dsl-textarea-background,
    var(--dsl-component-textarea-surface-background-default)
  );
  --_dsl-textarea-border: var(
    --dsl-textarea-border-color,
    var(--dsl-component-textarea-border-color-default)
  );
  --_dsl-textarea-color: var(
    --dsl-textarea-color,
    var(--dsl-component-textarea-content-color-default)
  );
  --_dsl-textarea-placeholder: var(
    --dsl-textarea-placeholder-color,
    var(--dsl-component-textarea-content-color-placeholder)
  );

  display: inline-flex;
  box-sizing: border-box;
  inline-size: var(--dsl-textarea-width, auto);
  block-size: var(--dsl-textarea-height, auto);
  vertical-align: middle;
  border: var(--dsl-component-textarea-border-width) solid var(--_dsl-textarea-border);
  border-radius: var(--dsl-component-textarea-border-radius);
  background: var(--_dsl-textarea-background);
  color: var(--_dsl-textarea-color);
  white-space: nowrap;

  &[data-fill] {
    inline-size: 100%;
  }

  > textarea {
    --_dsl-textarea-default-min-height: calc(
      var(--dsl-component-textarea-typography-line-height) +
        (var(--dsl-component-textarea-spacing-padding-block) * 2)
    );
    --_dsl-textarea-default-height: calc(
      (var(--dsl-component-textarea-typography-line-height) * 3) +
        (var(--dsl-component-textarea-border-width) * 2) +
        (var(--dsl-component-textarea-border-radius) / 2) +
        (var(--dsl-component-textarea-spacing-padding-block) * 2)
    );

    display: inline-block;
    box-sizing: border-box;
    inline-size: calc(
      var(--dsl-textarea-control-width, 100%) -
        (var(--dsl-component-textarea-border-width) * 2) -
        (var(--dsl-component-textarea-border-radius) / 2)
    );
    block-size: calc(
      var(--dsl-textarea-control-height, var(--_dsl-textarea-default-height)) -
        (var(--dsl-component-textarea-border-width) * 2) -
        (var(--dsl-component-textarea-border-radius) / 2)
    );
    min-block-size: var(--_dsl-textarea-default-min-height);
    margin-block-start: calc(var(--dsl-component-textarea-border-radius) / 4);
    margin-inline-end: calc(
      var(--dsl-component-textarea-resize-clearance) +
        (var(--dsl-component-textarea-border-radius) / 4)
    );
    margin-block-end: calc(
      var(--dsl-component-textarea-resize-clearance) +
        (var(--dsl-component-textarea-border-radius) / 4)
    );
    margin-inline-start: calc(var(--dsl-component-textarea-border-radius) / 4);
    padding-block: var(--dsl-component-textarea-spacing-padding-block);
    padding-inline: var(--dsl-component-textarea-spacing-padding-inline);
    vertical-align: middle;
    border: 0;
    outline: 0;
    background: transparent;
    color: inherit;
    font-family: var(--dsl-component-textarea-typography-family);
    font-size: var(--dsl-component-textarea-typography-size);
    font-weight: var(--dsl-component-textarea-typography-weight);
    line-height: var(--dsl-component-textarea-typography-line-height);
    resize: var(--dsl-textarea-resize, both);
  }

  &[data-fill] > textarea {
    resize: var(--dsl-textarea-resize, vertical);
  }

  > textarea::placeholder {
    color: var(--_dsl-textarea-placeholder);
    opacity: 1;
  }

  /*
   * Preserve native autofill behavior without exposing browser-specific fill
   * colors that can conflict with Concord themes.
   */
  > textarea:is(:autofill, :-webkit-autofill) {
    -webkit-text-fill-color: var(--_dsl-textarea-color);
    caret-color: var(--_dsl-textarea-color);
    -webkit-box-shadow: 0 0 0 1000px var(--_dsl-textarea-background) inset;
    box-shadow: 0 0 0 1000px var(--_dsl-textarea-background) inset;
  }

  &:hover:not(:has(> textarea:disabled)) {
    --_dsl-textarea-border: var(--dsl-component-textarea-border-color-hover);
  }

  &:has(> textarea:focus-visible) {
    --_dsl-textarea-border: var(--dsl-component-textarea-border-color-focus);
    outline: var(--dsl-component-textarea-focus-width)
      var(--dsl-component-textarea-focus-style) var(--dsl-component-textarea-focus-color);
    outline-offset: var(--dsl-component-textarea-focus-offset);
  }

  &:has(> textarea[aria-invalid]:not([aria-invalid='false'])) {
    --_dsl-textarea-border: var(--dsl-component-textarea-border-color-invalid);
  }

  &:has(> textarea:disabled) {
    --_dsl-textarea-background: var(--dsl-component-textarea-surface-background-disabled);
    --_dsl-textarea-border: var(--dsl-component-textarea-border-color-disabled);
    --_dsl-textarea-color: var(--dsl-component-textarea-content-color-disabled);
    --_dsl-textarea-placeholder: var(--dsl-component-textarea-content-color-disabled);
    cursor: not-allowed;
  }

  > textarea:disabled {
    resize: none;
    -webkit-text-fill-color: currentColor;
    opacity: 1;
  }
}

      color: GrayText;
    }
  }
}

Component contract

Accessibility

WCAG 2.2 AA target

Native semantics

  • Preserves native HTML semantics and browser behavior for the textarea.

Requirements

  • Every textarea requires an accessible name.
  • When external guidance, supporting text, or validation messages apply, associate them with the native textarea using aria-describedby.
