Form Label

Component implementation

packages/css/src/components/label.css 
.dsl-label {
  display: inline-block;

  color: var(--dsl-component-label-content-color-default);
  font-family: var(--dsl-component-label-typography-family);
  font-size: var(--dsl-component-label-typography-size);
  font-weight: var(--dsl-component-label-typography-weight-strong);
  line-height: var(--dsl-component-label-typography-line-height);

  > label {
    color: inherit;
    font: inherit;
    overflow-wrap: anywhere;
  }

  > .indicator {
    margin-inline-start: var(--dsl-component-label-spacing-gap);
    color: inherit;
    font-weight: var(--dsl-component-label-typography-weight-regular);
  }

  &[data-indicator='required'] > .indicator {
    color: var(--dsl-component-label-indicator-required);
  }

  &[data-tone='subtle'] {
    color: var(--dsl-component-label-content-color-subtle);
    font-weight: var(--dsl-component-label-typography-weight-regular);
  }

  > .annotation {
    margin-inline-start: var(--dsl-component-label-spacing-annotation-gap);
    padding: 0;
    vertical-align: middle;
    border: 0;
    background: transparent;
    color: var(--dsl-component-label-annotation-color);
    cursor: pointer;
  }

  > .annotation:focus-visible {
    outline: var(--dsl-semantic-focus-outline-width)
      var(--dsl-semantic-focus-outline-style) var(--dsl-semantic-focus-outline-color);
    outline-offset: var(--dsl-semantic-focus-outline-offset);
  }

  > .annotation > .dsl-icon {
    --dsl-icon-size: var(--dsl-component-label-annotation-size);
  }
}

@media (forced-colors: active) {
  .dsl-label {
    color: CanvasText;

    &[data-indicator='required'] > .indicator {
      color: MarkText;
      background: Mark;
    }

    > .annotation {
      color: ButtonText;
    }

    > .annotation:focus-visible {
      outline-color: Highlight;
    }
  }
}
