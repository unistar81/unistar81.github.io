.dsl-slider {
  --_dsl-slider-marker-width: calc(var(--dsl-component-slider-track-height) / 2);

  display: inline-block;
  inline-size: var(--dsl-slider-width, 100%);
  block-size: var(--dsl-slider-height, auto);
  color: var(--dsl-component-slider-content-color);
  font-family: var(--dsl-component-slider-typography-family);

  > .info,
  > .range {
    display: flex;
    align-items: center;
    justify-content: space-between;
  }

  > .info {
    margin-block-end: var(--dsl-component-slider-spacing-gap);
    font-size: var(--dsl-component-slider-typography-size);
    font-weight: var(--dsl-component-slider-typography-weight);
    line-height: var(--dsl-component-slider-typography-line-height);
  }

  > .info > label {
    min-inline-size: 0;
    overflow-wrap: anywhere;
  }

  > .info > output {
    flex: 0 0 auto;
    margin-inline-start: var(--dsl-component-slider-spacing-gap);
  }

  > .range {
    inline-size: 100%;
  }

  > .range > .track {
    flex: 1 1 auto;
    min-inline-size: 0;
  }

  > .range > .track > input[type='range'] {
    --_dsl-slider-progress: attr(data-progress type(<percentage>), 0%);
    appearance: none;
    inline-size: 100%;
    block-size: var(--dsl-component-slider-track-height);
    margin-inline: 0;
    margin-block: calc(
      (
          var(--dsl-component-slider-thumb-size) -
            var(--dsl-component-slider-track-height)
        ) /
        2
    );
    padding: 0;
    border: 0;
    border-radius: var(--dsl-component-slider-track-radius);
    outline: 0;
    background: linear-gradient(
      to right,
      var(--dsl-component-slider-track-color-active) 0 var(--_dsl-slider-progress),
      var(--dsl-component-slider-track-color-inactive) var(--_dsl-slider-progress) 100%
    );
    cursor: pointer;
  }

  > .range > .minimum {
    margin-inline-end: var(--dsl-component-slider-spacing-range-gap);
  }

  > .range > .maximum {
    overflow-wrap: normal;
    margin-inline-start: var(--dsl-component-slider-spacing-range-gap);
  }

  > .range > .track > input[type='range']::-webkit-slider-thumb {
    appearance: none;
    inline-size: var(--dsl-component-slider-thumb-size);
    block-size: var(--dsl-component-slider-thumb-size);
    border: var(--dsl-component-slider-thumb-border-width) solid
      var(--dsl-component-slider-thumb-border-color);
    border-radius: 50%;
    background: var(--dsl-component-slider-thumb-background);
  }

  > .range > .track > input[type='range']::-moz-range-thumb {
    inline-size: var(--dsl-component-slider-thumb-size);
    block-size: var(--dsl-component-slider-thumb-size);
    border: var(--dsl-component-slider-thumb-border-width) solid
      var(--dsl-component-slider-thumb-border-color);
    border-radius: 50%;
    background: var(--dsl-component-slider-thumb-background);
  }

  > .range > .track > input[type='range']:focus-visible::-webkit-slider-thumb {
    outline: var(--dsl-component-slider-focus-width)
      var(--dsl-component-slider-focus-style) var(--dsl-component-slider-focus-color);
    outline-offset: var(--dsl-component-slider-focus-offset);
  }

  > .range > .track > input[type='range']:focus-visible::-moz-range-thumb {
    outline: var(--dsl-component-slider-focus-width)
      var(--dsl-component-slider-focus-style) var(--dsl-component-slider-focus-color);
    outline-offset: var(--dsl-component-slider-focus-offset);
  }

  > .range > .track > input[type='range']:disabled {
    opacity: 0.55;
    cursor: not-allowed;
  }

  > .range > .track > .markers {
    display: none;
  }

  &:is([data-tick-marks], [data-range-labels], [data-markers], [data-marker-labels])
    > .range
    > .track
    > .markers:not([hidden]) {
    display: flex;
    justify-content: space-between;
    margin-block-start: var(--dsl-component-slider-spacing-marker-gap);
    margin-inline: calc(
      (var(--dsl-component-slider-thumb-size) / 2) - (var(--_dsl-slider-marker-width) / 2)
    );
  }

  &:is([data-tick-marks], [data-range-labels], [data-markers], [data-marker-labels])
    > .range
    > .track
    > .markers
    > span {
    display: inline-flex;
    flex: 0 0 var(--_dsl-slider-marker-width);
    flex-direction: column;
    align-items: center;
    justify-content: center;
    inline-size: var(--_dsl-slider-marker-width);

    color: var(--dsl-component-slider-marker-color);
    font-size: var(--dsl-component-slider-marker-typography-size);
    text-align: center;
    white-space: nowrap;
  }

  &:is([data-tick-marks], [data-markers]) > .range > .track > .markers > span::before {
    content: '';

    display: block;
    flex: 0 0 var(--dsl-component-slider-track-height);

    inline-size: var(--_dsl-slider-marker-width);
    block-size: var(--dsl-component-slider-track-height);
    margin-inline: auto;

    border-radius: var(--dsl-component-slider-track-radius);
    background: var(--dsl-component-slider-marker-color);
  }

  &:is([data-tick-marks], [data-markers]):is([data-range-labels], [data-marker-labels])
    > .range
    > .track
    > .markers
    > span::before {
    margin-block-end: var(--dsl-component-slider-spacing-marker-gap);
  }
}
