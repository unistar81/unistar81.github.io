Accessibility notes
  • Use role="status" for non-urgent updates and role="alert" only when information requires immediate attention.
  • Variant color and iconography supplement the message rather than replacing meaningful text.

packages/css/src/components/alert.css 
.dsl-alert {
  --_dsl-alert-accent: var(--dsl-component-alert-color-information);

  display: flex;
  align-items: flex-start;
  gap: var(--dsl-component-alert-spacing-gap);
  inline-size: var(--dsl-alert-width, 100%);
  padding: var(--dsl-component-alert-spacing-padding);
  border: var(--dsl-component-alert-border-width) solid var(--_dsl-alert-accent);
  border-radius: var(--dsl-component-alert-border-radius);
  background: var(--dsl-component-alert-surface-background);
  color: var(--dsl-component-alert-content-color);

  &[data-variant='success'] {
    --_dsl-alert-accent: var(--dsl-component-alert-color-success);
  }

  &[data-variant='warning'] {
    --_dsl-alert-accent: var(--dsl-component-alert-color-warning);
  }

  &[data-variant='critical'] {
    --_dsl-alert-accent: var(--dsl-component-alert-color-critical);
  }

  > .icon {
    --dsl-icon-color: var(--_dsl-alert-accent);
    --dsl-icon-size: var(--dsl-component-alert-icon-size);
    flex: 0 0 auto;
  }

  > .content {
    flex: 1 1 auto;
    min-inline-size: 0;
  }

  > .content > .title,
  > .content > .message {
    display: block;
    overflow-wrap: anywhere;
  }

  > .content > .title {
    font-size: var(--dsl-component-alert-title-typography-size);
    font-weight: var(--dsl-component-alert-title-typography-weight);
    line-height: var(--dsl-component-alert-title-typography-line-height);
  }

  > .content > .title + .message {
    margin-block-start: var(--dsl-component-alert-spacing-content-gap);
  }

  > .content > .message {
    font-size: var(--dsl-component-alert-message-typography-size);
    line-height: var(--dsl-component-alert-message-typography-line-height);
  }

  > .close {
    flex: 0 0 auto;
    padding: 0;
    border: 0;
    background: transparent;
    color: var(--_dsl-alert-accent);
    cursor: pointer;
  }

  > .close:focus-visible {
    outline: var(--dsl-semantic-focus-outline-width)
      var(--dsl-semantic-focus-outline-style) var(--dsl-semantic-focus-outline-color);
    outline-offset: var(--dsl-semantic-focus-outline-offset);
  }

  &[data-layout='inline'] {
    display: inline-block;
    inline-size: auto;
    padding: 0;
    border: 0;
    background: transparent;

    > .icon,
    > .content,
    > .content > .message {
      display: inline;
    }

    > .icon {
      margin-inline-end: var(--dsl-component-alert-spacing-content-gap);
      vertical-align: -0.125em;
    }
  }
}

@media (forced-colors: active) {
  .dsl-alert {
    border-color: CanvasText;

    > .icon,
    > .close {
      color: CanvasText;
    }
  }
}
