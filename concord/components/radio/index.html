packages/css/src/components/radio.css 
.dsl-radio {
  --_dsl-radio-background: var(--dsl-component-radio-control-background-default);
  --_dsl-radio-border: var(--dsl-component-radio-control-border-default);
  --_dsl-radio-dot: var(--dsl-component-radio-control-dot-selected);
  --_dsl-radio-label: var(--dsl-component-radio-content-label-color-default);
  --_dsl-radio-description: var(--dsl-component-radio-content-description-color-default);

  display: inline-block;
  inline-size: var(--dsl-radio-width, auto);
  white-space: nowrap;
  color: var(--_dsl-radio-label);
  cursor: pointer;

  > label {
    display: inline-flex;
    align-items: center;
    max-inline-size: 100%;
    font-size: 0;
    white-space: nowrap;
    cursor: inherit;
  }

  > label > input[type='radio'] {
    appearance: none;
    position: relative;
    display: block;
    flex: 0 0 auto;
    inline-size: var(--dsl-component-radio-control-size);
    block-size: var(--dsl-component-radio-control-size);
    margin: var(--dsl-component-radio-control-margin);
    padding: 0;
    border: var(--dsl-component-radio-control-border-width) solid var(--_dsl-radio-border);
    border-radius: 50%;
    outline: none;
    background: var(--_dsl-radio-background);
    color: var(--_dsl-radio-dot);
    cursor: inherit;
  }

  > label > input[type='radio']::before {
    content: '';
    position: absolute;
    inset: 50% auto auto 50%;
    inline-size: var(--dsl-component-radio-control-dot-size);
    block-size: var(--dsl-component-radio-control-dot-size);
    border-radius: 50%;
    background: currentColor;
    transform: translate(-50%, -50%) scale(0);
    transition: transform var(--dsl-semantic-motion-interaction-duration)
      var(--dsl-semantic-motion-interaction-easing);
  }

  > label > input[type='radio']:checked::before {
    transform: translate(-50%, -50%) scale(1);
  }

  > label > .content {
    display: block;
    min-inline-size: 0;
    max-inline-size: calc(
      100% -
        (
          var(--dsl-component-radio-control-size) +
            (var(--dsl-component-radio-control-margin) * 2)
        )
    );
    color: var(--_dsl-radio-label);
    font-family: var(--dsl-component-radio-content-label-typography-family);
    font-size: var(--dsl-component-radio-content-label-typography-size);
    line-height: var(--dsl-component-radio-content-label-typography-line-height);
    white-space: normal;
    overflow-wrap: anywhere;
  }

  > label > .content > .label,
  > label > .content > .description {
    display: block;
    white-space: normal;
  }

  > label > .content > .description {
    color: var(--_dsl-radio-description);
    font-family: var(--dsl-component-radio-content-description-typography-family);
    font-size: var(--dsl-component-radio-content-description-typography-size);
    line-height: var(--dsl-component-radio-content-description-typography-line-height);
  }

  &:hover > label > input[type='radio']:not(:disabled) {
    --_dsl-radio-border: var(--dsl-component-radio-control-border-hover);
  }

  > label > input[type='radio']:checked:where(:not(:disabled)) {
    --_dsl-radio-border: var(--dsl-component-radio-control-border-selected);
  }

  > label > input:focus-visible {
    outline: var(--dsl-component-radio-focus-width) var(--dsl-component-radio-focus-style)
      var(--dsl-component-radio-focus-color);
    outline-offset: var(--dsl-component-radio-focus-offset);
  }

  &:has(input:disabled) {
    --_dsl-radio-background: var(--dsl-component-radio-control-background-disabled);
    --_dsl-radio-border: var(--dsl-component-radio-control-border-disabled);
    --_dsl-radio-dot: var(--dsl-component-radio-control-dot-disabled);
    --_dsl-radio-label: var(--dsl-component-radio-content-label-color-disabled);
    --_dsl-radio-description: var(
      --dsl-component-radio-content-description-color-disabled
    );
    cursor: not-allowed;
  }
}
